\section{公开 API 速查（部署相关）}
\label{app:api}

下列为部署与整理相关入口摘要；完整签名以 \texttt{include/fcs.h}（聚合；声明在 \texttt{fcs\_store.h}）为准。

\begin{lstlisting}[language=fcsapi,caption={整理、Post 与养洞旋钮}]
void configure_deploy(bool async = true, u32 hw_pct = 150);
RC   polish();
void set_tidy_policy(TidyPolicy p);
TidySnap tidy_snap() const;
RC   maybe_tidy();

enum SoftCons { Off, Eager, Idle, Post };
void set_soft_cons(SoftCons m);       // Soft 默认 Post（坑 151）
SoftCons soft_cons() const;
RC   soft_cons_repay(u32 max_steps = 64);  // 读前还债

void set_part_hole_fit(bool on);
void set_part_frag(bool on);
void set_part_frag_floor(u32 n);
void set_part_coalesce(bool on);      // Soft 默认关
void set_protect_merge_holes(bool on);
void set_part_starve_fit(bool on);    // Soft 默认开（坑 144）
void set_starve_promote(bool on);     // Soft 默认开

void set_async_writes(bool on);
void set_merge_cheap_pages(u32 n);
void set_hw_reclaim_pct(u32 pct);
void set_space_accel_pct(u32 pct);    // Soft 默认 0
void set_maint_mode(int mode);        // 0 / 1 / 3
u64  durable_vers() const;
\end{lstlisting}

\begin{description}[leftmargin=1.4em]
  \item[\texttt{configure\_deploy}] Soft 一揽子：async、cheap、hole\_fit、frag~A、护洞、hw、\textbf{P2}、\textbf{Post}；必要时 maint $0\to 1$。
  \item[\texttt{soft\_cons\_repay}] Post 负债收口（consolidate$\to$relocate$\to$checkpoint$\to$shrink）；对外读前调用。
  \item[\texttt{polish}] 全库归一 + 尾部回落；写放大高，用于发布前。
  \item[\texttt{maybe\_tidy}] Soft+Post/Off 直接 OK；Eager/Idle 或 Hard/Periodic 按压力步进。
\end{description}
